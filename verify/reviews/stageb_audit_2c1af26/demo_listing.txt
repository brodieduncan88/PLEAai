#!/bin/bash
# /plea-init - PID 1 inside the Stage B microVM. Trusted guest component, pinned inside rootfs.ext4.
# The payload on /dev/vdb is untrusted: it runs as uid 10001 under `timeout`, never as root.
# Wire format: /dev/vdb = 8-byte big-endian length N (1..pcap) + N payload bytes.
#              /dev/vdc = 8-byte big-endian length L + L bytes; body = one JSON header line + stdout + stderr.
# Console: "PLEA_READY <nonce>" before the payload runs; "PLEA_DONE <nonce> <STATUS>" after the result is
# written. The nonce is drawn here from /dev/urandom after boot; the workload cannot read it (it is not in
# the kernel cmdline or the environment, and the workload has no read access to the console).

# BEGIN-LIB  (selftest.py sources this section on the build host; it has no side effects)
be64() {  # print an unsigned integer as 8 big-endian bytes
	local n=$1 i s=""
	for i in 56 48 40 32 24 16 8 0; do s+=$(printf '\\%03o' $(( (n >> i) & 255 ))); done
	printf "$s"
}
read_be64() {  # file -> echo the 8-byte big-endian prefix; fail if short or top bit set
	local b
	b=$(dd if="$1" bs=8 count=1 2>/dev/null | od -An -v -tu1) || return 1
	set -- $b
	[ $# -eq 8 ] || return 1
	[ "$1" -lt 128 ] || return 1
	local n=0 x
	for x in "$@"; do n=$(( (n << 8) | x )); done
	echo "$n"
}
extract_payload() {  # in_dev in_size pcap out_file -> sets PLEN; returns 2 on any malformed prefix
	local in=$1 insz=$2 pcap=$3 out=$4 n
	n=$(read_be64 "$in") || return 2
	[ "$n" -ge 1 ] && [ "$n" -le "$pcap" ] && [ $(( n + 8 )) -le "$insz" ] || return 2
	dd if="$in" of="$out" iflag=skip_bytes,count_bytes skip=8 count="$n" bs=65536 2>/dev/null || return 2
	[ "$(stat -c %s "$out")" = "$n" ] || return 2
	PLEN=$n
}
